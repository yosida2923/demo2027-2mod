#!/bin/bash
# Скрипт настройки Samba AD DC на ALT Linux
# Запускать от root: ./setup_samba.sh
# Требует предварительно настроенный домен Samba AD (samba-tool domain provision)

set -e  # остановка при ошибке

# ---- Проверка прав ----
if [ "$(id -u)" -ne 0 ]; then
    echo "Скрипт нужно запускать от root!" >&2
    exit 1
fi

# ---- Проверка, что samba-tool доступен и домен существует ----
if ! command -v samba-tool >/dev/null 2>&1; then
    echo "samba-tool не найден. Установите samba: apt-get install samba-DC" >&2
    exit 1
fi

# ---- Создание группы hq ----
echo ">>> Создание группы hq"
if samba-tool group list | grep -qw hq; then
    echo "Группа hq уже существует, пропускаем."
else
    samba-tool group add hq
fi

# ---- Создание пользователей ----
echo ">>> Создание пользователей hquser1..hquser5"
for i in $(seq 1 5); do
    USER="hquser$i"
    if samba-tool user list | grep -qw "$USER"; then
        echo "Пользователь $USER уже существует, пропускаем создание."
    else
        samba-tool user add "$USER" 'P@ssw0rd'
    fi

    echo "    -> Срок действия пароля $USER = бессрочно"
    samba-tool user setexpiry "$USER" --noexpiry

    echo "    -> Добавление $USER в группу hq"
    samba-tool group addmembers hq "$USER"
done

# ---- Политика паролей домена ----
echo ">>> Настройка политики паролей"
samba-tool domain passwordsettings set --min-pwd-length=8
samba-tool domain passwordsettings set --account-lockout-threshold=5

# ---- Срок действия администратора ----
echo ">>> Отключение срока действия для administrator"
samba-tool user setexpiry administrator --noexpiry

echo ">>> Готово."
