#!/bin/bash
# HQ-CLI setup script
# Запускать с правами root
# ВНИМАНИЕ: шаги 1-2 выполняются от root. После шага 2 обязательна перезагрузка.
# Шаги 3 и далее запускать после перезагрузки!

set -e

# ============================================================
# 1. Отключение автоматического DNS в NetworkManager
# ============================================================
CON_NAME="Проводное подключение 1"
# Проверяем, существует ли соединение с таким именем
if nmcli -t -f NAME connection show | grep -Fxq "$CON_NAME"; then
    nmcli connection modify "$CON_NAME" ipv4.ignore-auto-dns yes
    nmcli connection up "$CON_NAME" || true
    echo "[OK] DNS auto ignored for '$CON_NAME'"
else
    echo "[WARN] Connection '$CON_NAME' not found. Available connections:"
    nmcli -t -f NAME connection show
    echo "Please modify CON_NAME in the script and re-run."
fi

# ============================================================
# 2. Ввод в домен (автоматически)
# ============================================================
# Параметры домена (при необходимости 
# ============================================================
# 3. roleadd — выполнять ПОСЛЕ ввода в домен и перезагрузки
# ============================================================
roleadd hq wheel

# ============================================================
# 4. /etc/sudoers — точечная правка
# ============================================================
SUDOERS="/etc/sudoers"
BACKUP="/etc/sudoers.bak.$(date +%Y%m%d%H%M%S)"

cp -a "$SUDOERS" "$BACKUP"
echo "[OK] Backup: $BACKUP"

# --- 4.1 User_Alias WHEEL_USERS ---
# Ищем строку User_Alias WHEEL_USERS и заменяем её на нужную (или добавляем)
if grep -qE '^\s*#*\s*User_Alias\s+WHEEL_USERS\s*=' "$SUDOERS"; then
    sed -i -E 's|^\s*#*\s*User_Alias\s+WHEEL_USERS\s*=.*|User_Alias WHEEL_USERS = %wheel, %AU-TEAM\\\\hq|' "$SUDOERS"
else
    # Если нет, добавляем после последнего User_Alias
    awk '
        /^[[:space:]]*User_Alias[[:space:]]+/ { last=NR }
        { lines[NR]=$0 }
        END {
            for (i=1; i<=NR; i++) {
                print lines[i]
                if (i==last) print "User_Alias WHEEL_USERS = %wheel, %AU-TEAM\\\\hq"
            }
        }
    ' "$SUDOERS" > "${SUDOERS}.tmp" && mv "${SUDOERS}.tmp" "$SUDOERS"
fi

# --- 4.2 Cmnd_Alias SHELLCMD — вставить после 4-го Cmnd_Alias (PKGMAN) ---
NEW_CMND='Cmnd_Alias SHELLCMD = /usr/bin/id, /usr/bin/cat, /bin/grep'

# Удаляем старые определения SHELLCMD, если скрипт запускается повторно
sed -i -E '/^\s*#*\s*Cmnd_Alias\s+SHELLCMD\s*=/d' "$SUDOERS"

# Вставляем строго после 4-й строки, начинающейся с Cmnd_Alias
awk -v new="$NEW_CMND" '
    /^[[:space:]]*Cmnd_Alias[[:space:]]+/ {
        count++
        if (count == 4) print_after = 1
    }
    { print $0 }
    print_after { print new; print_after = 0 }
' "$SUDOERS" > "${SUDOERS}.tmp" && mv "${SUDOERS}.tmp" "$SUDOERS"

# --- 4.3 Строка правила для WHEEL_USERS (ТОЛЬКО ПЕРВАЯ) ---
# Раскомментируем и меняем ТОЛЬКО первую строку (верхнюю).
# Вторая строка ("Same thing without a password") останется закомментированной.
if grep -qE '^\s*#*\s*WHEEL_USERS\s+ALL\s*=' "$SUDOERS"; then
    # Ищем первую строку с WHEEL_USERS ALL=, игнорируя возможные комментарии,
    # и заменяем её. Остальные (вторую) не трогаем.
    sed -i '0,/^\s*#*\s*WHEEL_USERS\s\+ALL\s*=.*/s||WHEEL_USERS ALL=(ALL:ALL) SHELLCMD|' "$SUDOERS"
else
    # Если таких строк вообще нет — добавляем одну после User_Alias WHEEL_USERS
    sed -i -E "/^\s*User_Alias\s+WHEEL_USERS\s*=/a\\WHEEL_USERS ALL=(ALL:ALL) SHELLCMD" "$SUDOERS"
fi

# --- 4.4 Проверка синтаксиса ---
if visudo -cf "$SUDOERS"; then
    echo "[OK] /etc/sudoers updated and validated"
else
    echo "[ERROR] sudoers syntax error — restoring backup"
    cp -a "$BACKUP" "$SUDOERS"
    exit 1
fi

# ============================================================
# 5. Chrony — заменить или добавить строку pool
# ============================================================
CHRONY_CONF="/etc/chrony.conf"
# Удалить все старые строки server/pool, затем добавить нужную
sed -i '/^\s*\(server\|pool\)\s\+/d' "$CHRONY_CONF"
# Если после удаления файл не заканчивается переводом строки, добавить
[ -n "$(tail -c1 "$CHRONY_CONF")" ] && echo >> "$CHRONY_CONF"
echo "pool 172.16.1.1 iburst" >> "$CHRONY_CONF"

timedatectl set-ntp 0 || true
systemctl restart chronyd
systemctl enable chronyd
timedatectl set-ntp 1 || true
echo "[OK] Chrony configured with pool 172.16.1.1"

# ============================================================
# 6. SSH — установить Port 2027
# ============================================================
SSHD_CONF="/etc/openssh/sshd_config"
if [ -f "$SSHD_CONF" ]; then
    # Если строка Port существует (в любом виде), заменить её; иначе добавить
    if grep -qE '^\s*#*\s*Port\s+' "$SSHD_CONF"; then
        sed -i -E 's/^\s*#*\s*Port\s+.*/Port 2027/' "$SSHD_CONF"
    else
        echo "Port 2027" >> "$SSHD_CONF"
    fi
    systemctl restart sshd
    echo "[OK] sshd Port set to 2027"
else
    echo "[ERROR] $SSHD_CONF not found"
fi

# ============================================================
# 7. Установка Yandex Browser
# ============================================================
# В репозитории ALT Linux пакет называется yandex-browser-stable
if apt-get install -y yandex-browser-stable; then
    echo "[OK] yandex-browser-stable installed"
else
    echo "[WARN] yandex-browser-stable not found, trying yandex-browser"
    apt-get install -y yandex-browser || echo "[ERROR] Could not install Yandex Browser"
fi

# ============================================================
# 8. Показать информацию о дисках (из ваших заметок)
# ============================================================
cat /etc/fstab
df -h
