#!/bin/bash
# BR-SRV setup script
set -e

# ============================================================
# 1. Samba AD: группа и пользователи
# ============================================================
samba-tool group add hq 2>/dev/null || echo "[INFO] Группа hq уже существует"

for i in {1..5}; do
    samba-tool user add hquser$i P@ssw0rd 2>/dev/null || echo "[INFO] hquser$i уже существует"
    # ВАЖНО: двойной дефис --noexpiry (в заметках был длинный дефис –noexpiry)
    samba-tool user setexpiry --noexpiry hquser$i
    samba-tool group addmembers "hq" hquser$i
done

samba-tool domain passwordsettings set --min-pwd-length=8
samba-tool domain passwordsettings set --account-lockout-threshold=5
samba-tool user setexpiry --noexpiry administrator

# ============================================================
# 2. SSH: Port 2027 (путь в ALT Linux — /etc/openssh/sshd_config)
# ============================================================
SSHD_CONF="/etc/openssh/sshd_config"
if [ -f "$SSHD_CONF" ]; then
    if grep -qE '^\s*#*\s*Port\s+' "$SSHD_CONF"; then
        sed -i -E 's/^\s*#*\s*Port\s+.*/Port 2027/' "$SSHD_CONF"
    else
        echo "Port 2027" >> "$SSHD_CONF"
    fi
    systemctl restart sshd
    echo "[OK] sshd Port = 2027"
fi

# ============================================================
# 3. Chrony: заменить/добавить pool 172.16.2.1
# ============================================================
CHRONY_CONF="/etc/chrony.conf"
# Удалить все строки server/pool, затем добавить нужную
sed -i '/^\s*\(server\|pool\)\s\+/d' "$CHRONY_CONF"
[ -n "$(tail -c1 "$CHRONY_CONF")" ] && echo >> "$CHRONY_CONF"
echo "pool 172.16.2.1 iburst" >> "$CHRONY_CONF"

timedatectl set-ntp 0 || true
systemctl restart chronyd
systemctl enable chronyd
timedatectl set-ntp 1 || true
echo "[OK] Chrony pool = 172.16.2.1"

# ============================================================
# 4. Ansible: установка и настройка ansible.cfg
# ============================================================
apt-get install -y git ansible-core pip
pip install ansible-pylibssh

ANSIBLE_CFG="/etc/ansible/ansible.cfg"
if [ -f "$ANSIBLE_CFG" ]; then
    # Заменить или добавить remote_user = root в секции [defaults]
    if grep -qE '^\s*remote_user\s*=' "$ANSIBLE_CFG"; then
        sed -i -E 's/^\s*remote_user\s*=.*/remote_user = root/' "$ANSIBLE_CFG"
    else
        # Добавить после [defaults], если секция существует
        if grep -q '^\[defaults\]' "$ANSIBLE_CFG"; then
            sed -i '/^\[defaults\]/a remote_user = root' "$ANSIBLE_CFG"
        else
            echo -e "[defaults]\nremote_user = root" >> "$ANSIBLE_CFG"
        fi
    fi

    # Заменить или добавить interpreter_python = /usr/bin/python3
    if grep -qE '^\s*interpreter_python\s*=' "$ANSIBLE_CFG"; then
        sed -i -E 's|^\s*interpreter_python\s*=.*|interpreter_python = /usr/bin/python3|' "$ANSIBLE_CFG"
    else
        sed -i '/^\[defaults\]/a interpreter_python = /usr/bin/python3' "$ANSIBLE_CFG"
    fi
    echo "[OK] ansible.cfg обновлён"
fi

# ============================================================
# 5. SSH ключи и копирование на целевые хосты (интерактивно)
# ============================================================
if [ ! -f /root/.ssh/id_ed25519 ]; then
    ssh-keygen -t ed25519 -N "" -f /root/.ssh/id_ed25519
fi
# ssh-copy-id требует ввода пароля — выполните вручную или используйте sshpass
# ssh-copy-id -p 2027 root@hq-cli
# ssh-copy-id -p 2027 root@hq-srv

# ============================================================
# 6. Коллекции Ansible
# ============================================================
ansible-galaxy collection install git+https://github.com/ansible-collections/ansible.utils.git,v5.1.2
ansible-galaxy collection install git+https://github.com/ansible-collections/ansible.netcommon.git,v7.2.0
ansible-galaxy collection install git+https://github.com/ansible-collections/cisco.ios.git,v9.1.2

# ============================================================
# 7. /etc/ansible/hosts — ПОЛНАЯ ЗАМЕНА файла
# ============================================================
cat > /etc/ansible/hosts <<'EOF'
[hq]
hq-srv ansible_host=10.10.100.2 ansible_user=root ansible_password=toor ansible_port=2027
hq-cli ansible_host=10.10.200.2 ansible_user=root ansible_password=toor ansible_port=2027
hq-rtr ansible_host=172.16.1.2 ansible_user=admin ansible_password=admin ansible_become_password=admin ansible_connection=network_cli ansible_network_os=ios

[br]
br-rtr ansible_host=172.16.2.1 ansible_user=admin ansible_password=admin ansible_become_password=admin ansible_connection=network_cli ansible_network_os=ios
#bq-cli ansible_host=10.20.30.2
EOF
echo "[OK] /etc/ansible/hosts заменён"
